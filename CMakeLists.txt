cmake_minimum_required(VERSION 3.21)
project(scene-tree VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
add_library(scene-tree MODULE src/plugin.cpp)
target_link_libraries(scene-tree PRIVATE Qt6::Widgets OBS::libobs OBS::obs-frontend-api)
set_target_properties(scene-tree PROPERTIES PREFIX "")
install(TARGETS scene-tree LIBRARY DESTINATION bin/64bit)
include(CTest)
if(BUILD_TESTING)
  add_executable(row-test tests/row-test.cpp)
  target_link_libraries(row-test PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME row-test COMMAND row-test)
  set_tests_properties(row-test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
