cmake_minimum_required(VERSION 3.21)
project(unscene VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test Svg)
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
add_library(unscene MODULE src/plugin.cpp)
target_link_libraries(unscene PRIVATE Qt6::Widgets Qt6::Svg OBS::libobs OBS::obs-frontend-api)
set_target_properties(unscene PROPERTIES PREFIX "")
install(TARGETS unscene LIBRARY DESTINATION bin/64bit)
include(CTest)
if(BUILD_TESTING)
  add_executable(appearance-test tests/appearance-test.cpp)
  target_link_libraries(appearance-test PRIVATE Qt6::Widgets Qt6::Test Qt6::Svg)
  add_test(NAME appearance-test COMMAND appearance-test)
  set_tests_properties(appearance-test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(row-test tests/row-test.cpp)
  target_link_libraries(row-test PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME row-test COMMAND row-test)
  set_tests_properties(row-test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
