cmake_minimum_required(VERSION 3.21)
project(unscene VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test Svg)
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
add_library(unscene MODULE src/plugin.cpp)
target_link_libraries(unscene PRIVATE Qt6::Widgets Qt6::Svg OBS::libobs OBS::obs-frontend-api)
set_target_properties(unscene PROPERTIES PREFIX "")
install(TARGETS unscene LIBRARY DESTINATION bin/64bit)
include(CTest)
if(BUILD_TESTING)
  add_executable(appearance-test tests/appearance-test.cpp)
  target_link_libraries(appearance-test PRIVATE Qt6::Widgets Qt6::Test Qt6::Svg)
  add_test(NAME appearance-test COMMAND appearance-test)
  set_tests_properties(appearance-test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(theme-test tests/theme-test.cpp)
  target_link_libraries(theme-test PRIVATE Qt6::Widgets Qt6::Test Qt6::Svg)
  add_test(NAME theme-test COMMAND theme-test)
  set_tests_properties(theme-test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(navigation-test tests/navigation-test.cpp)
  target_link_libraries(navigation-test PRIVATE Qt6::Core)
  add_test(NAME navigation-test COMMAND navigation-test)
  add_executable(source-picker-test tests/source-picker-test.cpp)
  target_link_libraries(source-picker-test PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME source-picker-test COMMAND source-picker-test)
  set_tests_properties(source-picker-test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(native-menu-test tests/native-menu-test.cpp)
  target_link_libraries(native-menu-test PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME native-menu-test COMMAND native-menu-test)
