cmake_minimum_required(VERSION 3.21)
project(unscene VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
set(UNSCENE_PACKAGE_LAYOUT "auto" CACHE STRING "OBS package layout: auto, legacy, modern")
set_property(CACHE UNSCENE_PACKAGE_LAYOUT PROPERTY STRINGS auto legacy modern)
if(NOT UNSCENE_PACKAGE_LAYOUT MATCHES "^(auto|legacy|modern)$")
  message(FATAL_ERROR "Unknown UNSCENE_PACKAGE_LAYOUT")
endif()
if(WIN32)
  set(_test_qpa minimal)
else()
  set(_test_qpa offscreen)
endif()
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
add_library(unscene MODULE src/plugin.cpp)
target_link_libraries(unscene PRIVATE Qt6::Widgets Qt6::Svg OBS::libobs OBS::obs-frontend-api)
set_target_properties(unscene PROPERTIES PREFIX "")
if(APPLE)
  set_target_properties(unscene PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin
    MACOSX_BUNDLE_GUI_IDENTIFIER "com.myrqyry.unscene"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/PluginInfo.plist.in"
    INSTALL_RPATH "@executable_path/../Frameworks" INSTALL_RPATH_USE_LINK_PATH FALSE)
  target_sources(unscene PRIVATE LICENSE)
  set_source_files_properties(LICENSE PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  install(TARGETS unscene BUNDLE DESTINATION . LIBRARY DESTINATION .)
else()
  set(_module_destination "bin/64bit")
  if(UNSCENE_PACKAGE_LAYOUT STREQUAL "modern" OR
     (UNSCENE_PACKAGE_LAYOUT STREQUAL "auto" AND libobs_VERSION VERSION_GREATER_EQUAL "33.0"))
    set(_module_destination ".")
  endif()
  install(TARGETS unscene LIBRARY DESTINATION "${_module_destination}" RUNTIME DESTINATION "${_module_destination}")
  install(FILES LICENSE DESTINATION .)
endif()
